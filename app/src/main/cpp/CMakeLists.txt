cmake_minimum_required(VERSION 3.22.1)
project(hfp_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(APM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../apm")
set(APM_LIB "${CMAKE_CURRENT_LIST_DIR}/../jniLibs/${ANDROID_ABI}/libwebrtc-audio-processing-3.so")
if(NOT EXISTS "${APM_ROOT}/include/webrtc-audio-processing-3/api/audio/audio_processing.h")
    message(FATAL_ERROR "WebRTC APM headers missing. Run PREPARE_APM_WINDOWS.ps1 before building.")
endif()
if(NOT EXISTS "${APM_LIB}")
    message(FATAL_ERROR "WebRTC arm64 library missing. Run PREPARE_APM_WINDOWS.ps1 before building.")
endif()
add_library(webrtc_apm SHARED IMPORTED)
set_target_properties(webrtc_apm PROPERTIES IMPORTED_LOCATION "${APM_LIB}")
add_library(hfp_audio SHARED native_audio.cpp sound_engine.cpp)
target_include_directories(hfp_audio PRIVATE
    "${APM_ROOT}/include"
    "${APM_ROOT}/include/webrtc-audio-processing-3")
target_compile_definitions(hfp_audio PRIVATE WEBRTC_POSIX=1)
target_compile_options(hfp_audio PRIVATE -Wall -Wextra)
find_library(log-lib log)
find_library(aaudio-lib aaudio)
find_library(dl-lib dl)
target_link_libraries(hfp_audio PRIVATE webrtc_apm ${aaudio-lib} ${log-lib} ${dl-lib})
